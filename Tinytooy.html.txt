function update(){

}

function updatePlayer(){
  const p=player;
  if(p.dead){p.deathT++;if(p.deathT>60){state=ST.GAMEOVER;stopMusic();SFX.over()}return}
  if(p.invuln>0)p.invuln--;
  const left=kd('ArrowLeft')||kd('KeyA'),right=kd('ArrowRight')||kd('KeyD');
  const jumpK=kd('Space')||kd('ArrowUp')||kd('KeyW');
  const ice=p.onGround&&curPlatType()==='ice';
  const acc=ice?.18:.9,fri=ice?.96:.72;
  if(left&&!right){p.vx-=acc;p.facing=-1}else if(right&&!left){p.vx+=acc;p.facing=1}else{p.vx*=fri;if(Math.abs(p.vx)<.1)p.vx=0}
  p.vx=Math.max(-RUN,Math.min(RUN,p.vx));
  if(p.coyote>0)p.coyote--;if(p.jumpBuf>0)p.jumpBuf--;
  if(jumpK&&!p.prevJump)p.jumpBuf=JUMPBUF;p.prevJump=jumpK;
  if(p.jumpBuf>0&&(p.onGround||p.coyote>0)){
    p.vy=JUMP;p.onGround=false;p.coyote=0;p.jumpBuf=0;p.stretch=1.35;p.squash=.7;p.jumping=true;SFX.jump();
    spawnParticles(p.x+p.w/2,p.y+p.h,6,'#fff')}
  if(!jumpK&&p.vy<-6)p.vy*=.85;
  p.vy+=GRAV;if(p.vy>MAXFALL)p.vy=MAXFALL;
  p.x+=p.vx;
  for(const pl of allPlat()){if(pl.type==='cloud'||pl.type==='ice')continue;
    if(rectOv(p,pl)){if(p.vx>0)p.x=pl.x-p.w;else if(p.vx<0)p.x=pl.x+pl.w;p.vx=0}}
  p.y+=p.vy;p.onGround=false;
  for(const pl of allPlat()){
    if(rectOv(p,pl)){
      if(p.vy>=0&&p.prevY+p.h<=pl.y+8){p.y=pl.y-p.h;p.vy=0;p.onGround=true;
        if(p.jumping){p.squash=1.25;p.stretch=.85;p.jumping=false}}
      else if(p.vy<0&&pl.type!=='cloud'&&pl.type!=='ice'){p.y=pl.y+pl.h;p.vy=0}
    }
  }
  for(const mp of level.movingPlatforms)
    if(p.onGround&&p.x+p.w>mp.x&&p.x<mp.x+mp.w&&Math.abs(p.y+p.h-mp.y)<8)p.y=mp.y-p.h;

  if(p.y>VH+200)killPlayer(true);
  p.squash+=(1-p.squash)*.2;p.stretch+=(1-p.stretch)*.2;p.runAnim+=Math.abs(p.vx)*.15;

function updateEnemies(){
  for(const e of level.enemies){
    if(!e.alive)continue;e.anim=(e.anim||0)+.15;
    if(e.type==='walker'){e.x+=e.vx;if(e.x<e.minX||e.x+e.w>e.maxX)e.vx*=-1}
    else{e.timer=(e.timer||0)+1;e.x+=e.vx;if(e.x<e.minX||e.x+e.w>e.maxX)e.vx*=-1;
      e.vy=(e.vy||0)+GRAV;e.y+=e.vy;
      if(e.y+e.h>500){e.y=500-e.h;e.vy=0;if(e.timer>60){e.vy=-11;e.timer=0}}}
    if(rectOv(player,e)&&!player.dead){
      if(player.vy>0&&player.prevY+player.h<=e.y+12){
        e.alive=false;player.vy=-11;player.stretch=1.4;player.squash=.7;
        spawnParticles(e.x+e.w/2,e.y+e.h/2,12,'#ffcc00');
        addFloater(e.x+e.w/2,e.y,'+50','#ffd700');score+=50;SFX.stomp()
      } else damagePlayer()
    }
  }
}
function updateFlyers(){
  for(const f of level.flyers){
    if(!f.alive)continue;f.anim+=.2;f.phase+=f.speed;f.y=f.baseY+Math.sin(f.phase)*f.range;
    f.x+=Math.cos(f.phase*.5)*.8;
    if(rectOv(player,f)&&!player.dead){
      if(player.vy>0&&player.prevY+player.h<=f.y+10){
        f.alive=false;player.vy=-12;player.stretch=1.4;
        spawnParticles(f.x+f.w/2,f.y+f.h/2,12,'#88ddff');score+=60;SFX.stomp()
      } else damagePlayer()
    }
  }
}
function updateProjectiles(){
  for(let i=level.projectiles.length-1;i>=0;i--){
    const p=level.projectiles[i];p.x+=p.vx;p.y+=p.vy;p.life--;

function updateBoss(){
  const b=level.boss;if(!b||!b.alive)return;
  b.t++;if(b.hurt>0)b.hurt--;
  if(b.hp<=b.maxHp/2&&b.phase===1){b.phase=2;b.vx=-2.8;SFX.boss();cam.shake=8;cam.shakeT=30}
  b.x+=b.vx;if(b.x<100||b.x+b.w>level.width-100)b.vx*=-1;
  b.shotT--;
  if(b.shotT<=0){
    b.shotT=b.phase===1?120:70;
    const cx=b.x+b.w/2,cy=b.y+b.h/2;
    const dx=player.x+player.w/2-cx,dy=player.y+player.h/2-cy,d=Math.hypot(dx,dy)||1;
    const sp=b.phase===1?5:6.5;
    if(b.phase===1)level.projectiles.push({x:cx-10,y:cy-10,w:20,h:20,vx:dx/d*sp,vy:dy/d*sp,life:180});
    else for(let a=-.3;a<=.3;a+=.3){
      const co=Math.cos(a),si=Math.sin(a);
      level.projectiles.push({x:cx-10,y:cy-10,w:18,h:18,vx:(dx/d*co-dy/d*si)*sp,vy:(dx/d*si+dy/d*co)*sp,life:180});
    }
    SFX.boss();
  }
  if(rectOv(player,b)&&!player.dead){
    if(player.vy>0&&player.prevY+player.h<=b.y+20){
      b.hp--;b.hurt=20;player.vy=-13;player.stretch=1.5;
      cam.shake=6;cam.shakeT=15;spawnParticles(b.x+b.w/2,b.y+b.h/2,16,'#ff2222');SFX.stomp();
      if(b.hp<=0){b.alive=false;SFX.win()}
    } else damagePlayer()
  }
}

function wrap(text,maxW){
  const words=text.split(' '),lines=[];let line='';
  for(const w of words){const t=line?line+' '+w:w;
    if(ctx.measureText(t).width>maxW&&line){lines.push(line);line=w}else line=t}
  if(line)lines.push(line);return lines;
}
function shade(hex,amt){
  const c=hex.replace('#',''),n=parseInt(c,16);
  let r=(n>>16)+amt,g=((n>>8)&255)+amt,b=(n&255)+amt;
  r=Math.max(0,Math.min(255,r));g=Math.max(0,Math.min(255,g));b=Math.max(0,Math.min(255,b));
  return '#'+((r<<16)|(g<<8)|b).toString(16).padStart(6,'0');
}
